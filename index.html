Dragon for Bloxd

A flying dragon boss for World Code: it circles in orbit, lands in the center, bites, heals from crystals and leaves an egg when killed.

How to install

  1. Open your world in Bloxd and go to the World Code editor (world owner only).
  2. Click "Copy code" below and paste the whole script into the editor (delete any old code first).
  3. Save. Stand where you want the center of the orbit and type !dragon in chat.

The script is about 11,500 characters, well under the 16,000 character World Code limit.

Commands

CommandWhat it does
!dragonSpawns the dragon. The orbit center is where you stand; it flies 8 blocks above you.
!dragonkillRemoves the dragon with no reward and no egg.

Commands start with "!" because "/" does not work for chat scripts.

How it behaves

  • Orbit: circles at a constant height around the center, radius 20 blocks.
  • Bite: bites and knocks back anyone who gets close (12 damage). Creative players take no damage but still get knocked back.
  • Landing: every 30-60 seconds it lands in the center, a shockwave knocks back nearby players (4 damage), it rests for 6-9 seconds, then takes off. This is the best time to hit it.
  • Fighting: left-click while looking at it (up to 12 blocks away). Hands and items deal 5 damage, swords deal 10. It has 150 HP and a health bar is shown in chat.
  • Healing: while at least one Crystal block is within 50 blocks, the dragon heals 40 HP per second. Break the crystals first.
  • Crystals: breaking one makes an explosion with a sound, knocks back nearby players and deals 1 damage.
  • Death: the dragon falls and disappears, leaves an egg made of blocks, and the killer receives 500 Aura XP Orbs.

Settings (constants at the top of the code)

ConstantWhat it changes
ORBIT_RADIUS, ORBIT_HEIGHT, ORBIT_SPEEDOrbit radius, flight height above the spawn point, flight speed.
ORBIT_DIRCircling direction: 1 or -1.
ROT_SIGNIf the dragon flies sideways or tail-first, set it to -1.
LAND_MIN, LAND_MAXHow often it lands, in ticks (30 ticks = 1 second).
GROUND_MIN, GROUND_MAXHow long it rests on the ground, in ticks.
BITE_DAMAGE, BITE_RANGEBite damage and distance.
MAX_HPDragon health.
HIT_DAMAGE, SWORD_DAMAGE, HIT_REACHDamage per hit and hit distance.
HEAL_AMOUNT, HEAL_SECONDS, HEAL_RANGEHow much HP it heals, over how many seconds, and the radius it searches for crystals.
CRYSTALA word that must appear in the crystal block name.
ORB_ITEM, ORB_AMOUNTReward item and amount given to the killer.
EGG_BLOCKSBlock used for the egg (the first one in the list that can actually be placed).

Troubleshooting

  • Dragon does not heal: make sure the block name contains the word from CRYSTAL. Crystals are found automatically after spawning, which takes about 10 seconds.
  • No reward received: make sure you have inventory space: it is 8 slots of orbs.
  • Cannot hit the dragon: get within 12 blocks and aim at it, hits use your line of sight.
  • Errors in chat: the script prints "Dragon error: ..." so the problem can be traced.

World Code script

const PARTS = [
  [{blockName:"Black Concrete",size:[3.25,0.5,1.5]},[0.5,0,0],[0,0,0],"BloxdBlock"],
  [{blockName:"Black Concrete",size:[1.25,0.5,2]},[1.25,0,-2],[0,0,0],"BloxdBlock"],
  [{blockName:"Black Concrete",size:[1.25,0.5,2]},[1.25,0,2],[0,0,0],"BloxdBlock"],
  [{blockName:"Black Concrete",size:[0.75,0.5,2.5]},[-2.5,0,0],[0,-1.5708,0],"BloxdBlock"],
  [{blockName:"Black Concrete",size:[0.75,1,0.75]},[2.625,0,0],[0,-1.5708,0],"BloxdBlock"],
  [{blockName:"Black Concrete",size:[0.75,0.25,0.75]},[3.375,0.5,0],[0,-1.5708,0],"BloxdBlock"],
  [{blockName:"Black Concrete",size:[0.75,0.25,0.75]},[3.375,0,0],[0,-1.5708,0],"BloxdBlock"],
  [{blockName:"Obsidian",size:[1,0.25,0.75]},[3.25,0.25,0],[0,0,0],"BloxdBlock"],
  [{blockName:"Block of Quartz",size:[0.02,0.25,0.25]},[3.01,0.75,-0.25],[0,0,0],"BloxdBlock"],
  [{blockName:"Block of Quartz",size:[0.02,0.25,0.25]},[3.01,0.75,0.25],[0,0,0],"BloxdBlock"],
];

const ORBIT_RADIUS = 20, ORBIT_HEIGHT = 8, ORBIT_SPEED = 0.3;
const ORBIT_DIR = 1;
const ROT_SIGN = 1;

const LAND_MIN = 900, LAND_MAX = 1800;
const LAND_TICKS = 100, TAKEOFF_TICKS = 90;
const GROUND_MIN = 180, GROUND_MAX = 270;
const SHOCK_RADIUS = 8, SHOCK_DAMAGE = 4, SHOCK_KNOCK = 14;

const BITE_RANGE = 6, BITE_DAMAGE = 12, BITE_COOLDOWN = 20, KNOCK_H = 18, KNOCK_V = 12;

const MAX_HP = 150, HIT_DAMAGE = 5, SWORD_DAMAGE = 10;
const HIT_REACH = 12, HIT_RADIUS = 3.5, HIT_COOLDOWN = 6;
const ORB_ITEM = "Aura XP Orb", ORB_AMOUNT = 500;
const EGG_BLOCKS = ["Glowing Mushroom", "Glowstone", "Block of Quartz"];

const CRYSTAL = "Crystal";
const HEAL_RANGE = 50;
const HEAL_AMOUNT = 40, HEAL_SECONDS = 1;
const HEAL_STEP = HEAL_AMOUNT / (HEAL_SECONDS * 3);
const BLAST_RADIUS = 5, BLAST_DAMAGE = 1, BLAST_KNOCK = 10, BLAST_SOUND = "explode";
const SCAN_BUDGET = 3000;

let dragon = null, scan = null;
const crystals = {};

function hurt(pid, amount) {
  try {
    if (api.getHealth(pid) == null) return false;
    api.applyHealthChange(pid, -amount);
    return true;
  } catch (e) { return false; }
}
function knock(pid, x, y, z) { try { api.applyImpulse(pid, x, y, z); } catch (e) {} }
function rnd(a, b) { return a + Math.random() * (b - a); }
function ease(t) { return t * t * (3 - 2 * t); }
function lerpAngle(a, b, t) {
  let d = b - a;
  while (d > Math.PI) d -= Math.PI * 2;
  while (d < -Math.PI) d += Math.PI * 2;
  return a + d * t;
}

function blockAt(x, y, z) {
  let b;
  try { b = api.getBlock(x, y, z); } catch (e) {}
  if (typeof b !== "string") { try { b = api.getBlock([x, y, z]); } catch (e) {} }
  return b;
}
function isCrystal(name) { return typeof name === "string" && name.indexOf(CRYSTAL) !== -1; }
function groundAt(x, y, z, fb) {
  const bx = Math.floor(x), bz = Math.floor(z);
  for (let yy = Math.floor(y); yy > y - 80; yy--) if (api.getBlockSolidity(bx, yy, bz)) return yy + 1;
  return fb;
}

function giveReward(pid) {
  try {
    for (let left = ORB_AMOUNT; left > 0; left -= 64) api.giveItem(pid, ORB_ITEM, Math.min(64, left));
    api.sendMessage(pid, "Reward: " + ORB_AMOUNT + " x " + ORB_ITEM);
  } catch (e) { api.sendMessage(pid, "Could not give item " + ORB_ITEM); }
}

function placeEgg(d) {
  const bx = Math.floor(d.x), bz = Math.floor(d.z), by = Math.floor(d.ground);
  let name = null;
  for (const n of EGG_BLOCKS) {
    try { api.setBlock(bx, by, bz, n); if (blockAt(bx, by, bz) === n) { name = n; break; } } catch (e) {}
  }
  if (!name) return null;
  for (let dy = 0; dy <= 5; dy++) {
    const w = 2 + (2.5 - dy) * 0.2, h = (dy - 2.5) / 3;
    for (let dx = -3; dx <= 3; dx++) for (let dz = -3; dz <= 3; dz++) {
      if ((dx * dx + dz * dz) / (w * w) + h * h <= 1) { try { api.setBlock(bx + dx, by + dy, bz + dz, name); } catch (e) {} }
    }
  }
  return name;
}

function spawnDragon(pos) {
  const ids = [];
  for (const [opts, , , type] of PARTS) {
    const id = api.attemptCreateMeshEntity(type, opts);
    if (id == null) { for (const i of ids) api.deleteMeshEntity(i); return null; }
    ids.push(id);
  }
  const d = { ids, cx: pos[0], cy: pos[1] + ORBIT_HEIGHT, cz: pos[2], a: 0, x: 0, y: 0, z: 0,
              yaw: 0, age: 0, bite: 0, hp: MAX_HP, hitCd: 0, dying: 0, killer: null, ground: pos[1],
              st: 0, t: 0, nextLand: rnd(LAND_MIN, LAND_MAX) };
  place(d);
  const W = HEAL_RANGE * 2 + 1;
  scan = { x0: Math.floor(pos[0]) - HEAL_RANGE, z0: Math.floor(pos[2]) - HEAL_RANGE,
           y0: Math.floor(pos[1]) - 30, W, H: 61, i: 0, total: W * W * 61 };
  return d;
}

function orbitXZ(d, a) { return [d.cx + Math.cos(a) * ORBIT_RADIUS, d.cz + Math.sin(a) * ORBIT_RADIUS]; }
function tangentYaw(a) { return Math.atan2(-Math.cos(a) * ORBIT_DIR, -Math.sin(a) * ORBIT_DIR); }

function place(d) {
  const p = orbitXZ(d, d.a);
  d.x = p[0]; d.z = p[1];
  d.y = d.cy + Math.sin(d.age * 0.1) * 0.3;
  d.yaw = tangentYaw(d.a);
}

function startLanding(d) {
  d.gy = groundAt(d.cx, d.cy, d.cz, d.ground) + 0.6;
  d.sx = d.x; d.sy = d.y; d.sz = d.z; d.syaw = d.yaw;
  d.tyaw = Math.atan2(-(d.cz - d.z), d.cx - d.x);
  d.st = 1; d.t = 0;
}

function shockwave(d) {
  try { api.broadcastSound(BLAST_SOUND, 1, 1, { x: d.cx, y: d.gy, z: d.cz }); } catch (e) {}
  for (const pid of api.getPlayerIds()) {
    const p = api.getPosition(pid);
    if (!p) continue;
    const dx = p[0] - d.cx, dz = p[2] - d.cz, dist = Math.hypot(dx, p[1] - d.gy, dz);
    if (dist > SHOCK_RADIUS) continue;
    const k = SHOCK_KNOCK / Math.max(dist, 1);
    hurt(pid, SHOCK_DAMAGE);
    knock(pid, dx * k, 8, dz * k);
  }
}

function flyStep(d) {
  d.t++;
  if (d.st === 1) {
    const u = ease(Math.min(1, d.t / LAND_TICKS));
    d.x = d.sx + (d.cx - d.sx) * u; d.z = d.sz + (d.cz - d.sz) * u; d.y = d.sy + (d.gy - d.sy) * u;
    d.yaw = lerpAngle(d.syaw, d.tyaw, Math.min(1, d.t / 30));
    if (d.t >= LAND_TICKS) { d.st = 2; d.t = rnd(GROUND_MIN, GROUND_MAX); d.gyaw = d.yaw; shockwave(d); }
  } else {
    const e = orbitXZ(d, d.a), u = ease(Math.min(1, d.t / TAKEOFF_TICKS));
    d.x = d.cx + (e[0] - d.cx) * u; d.z = d.cz + (e[1] - d.cz) * u; d.y = d.gy + (d.cy - d.gy) * u;
    const trav = Math.atan2(-(e[1] - d.cz), e[0] - d.cx);
    const y1 = lerpAngle(d.gyaw, trav, Math.min(1, d.t / 30));
    d.yaw = lerpAngle(y1, tangentYaw(d.a), Math.max(0, (d.t / TAKEOFF_TICKS - 0.6) / 0.4));
    if (d.t >= TAKEOFF_TICKS) { d.st = 0; d.nextLand = rnd(LAND_MIN, LAND_MAX); place(d); }
  }
}

function move(d) {
  if (d.bite > 0) d.bite--;
  if (d.hitCd > 0) d.hitCd--;
  if (d.st === 0) {
    d.a += ORBIT_DIR * ORBIT_SPEED / ORBIT_RADIUS;
    place(d);
    if (--d.nextLand <= 0) startLanding(d);
  } else if (d.st === 2) {
    if (--d.t <= 0) { d.st = 3; d.t = 0; }
  } else flyStep(d);

  if (d.bite > 0 || d.age % 3 !== 0) return;
  for (const pid of api.getPlayerIds()) {
    const p = api.getPosition(pid);
    if (p && Math.hypot(p[0] - d.x, p[1] - d.y, p[2] - d.z) < BITE_RANGE) {
      hurt(pid, BITE_DAMAGE);
      knock(pid, Math.cos(d.yaw) * KNOCK_H, KNOCK_V, -Math.sin(d.yaw) * KNOCK_H);
      d.bite = BITE_COOLDOWN;
      break;
    }
  }
}

function render(d) {
  const c = Math.cos(d.yaw), s = Math.sin(d.yaw);
  const flap = d.st === 2 ? Math.sin(d.age * 0.06) * 0.1 : Math.sin(d.age * (d.dying ? 0.1 : 0.25));
  for (let i = 0; i < PARTS.length; i++) {
    const [, p, r] = PARTS[i];
    let py = p[1], pz = p[2];
    if (i === 1 || i === 2) py += flap * 0.9;
    else if (i === 3) pz += Math.sin(d.age * 0.15) * 0.3;
    api.setPosition(d.ids[i], d.x + p[0] * c + pz * s, d.y + py, d.z - p[0] * s + pz * c);
    api.setEntityRotation(d.ids[i], r[0], r[1] + d.yaw * ROT_SIGN, r[2]);
  }
}

function removeDragon() {
  for (const id of dragon.ids) api.deleteMeshEntity(id);
  dragon = null; scan = null;
}

function dieStep(d) {
  d.yaw += 0.4;
  d.y = Math.max(d.ground + 0.5, d.y - 0.5);
  render(d);
  if (--d.dying <= 0) {
    const egg = placeEgg(d);
    api.broadcastMessage("Dragon defeated!" + (egg ? " It left an egg (" + egg + ")." : ""));
    if (d.killer) giveReward(d.killer);
    removeDragon();
  }
}

function heal(d) {
  if (d.hp >= MAX_HP) return;
  const r2 = HEAL_RANGE * HEAL_RANGE;
  for (const key in crystals) {
    const c = crystals[key];
    const dx = c[0] - d.x, dy = c[1] - d.y, dz = c[2] - d.z;
    if (dx * dx + dy * dy + dz * dz > r2) continue;
    const b = blockAt(c[0], c[1], c[2]);
    if (typeof b === "string" && !isCrystal(b)) { delete crystals[key]; continue; }
    d.hp = Math.min(MAX_HP, d.hp + HEAL_STEP);
    return;
  }
}

function scanStep() {
  const s = scan;
  for (let n = 0; n < SCAN_BUDGET && s.i < s.total; n++) {
    const i = s.i++, y = s.y0 + (i % s.H), r = (i / s.H) | 0;
    const x = s.x0 + (r % s.W), z = s.z0 + ((r / s.W) | 0);
    if (isCrystal(blockAt(x, y, z))) crystals[x + "," + y + "," + z] = [x, y, z];
  }
  if (s.i >= s.total) scan = null;
}

tick = () => {
  if (scan) { try { scanStep(); } catch (e) { scan = null; } }
  if (!dragon) return;
  dragon.age++;
  try {
    if (dragon.dying > 0) { dieStep(dragon); return; }
    if (dragon.age % 10 === 0) heal(dragon);
    move(dragon);
    if (dragon.age % 2 === 0) render(dragon);
  } catch (e) { api.broadcastMessage("Dragon error: " + String(e && e.message || e)); }
};

onPlayerClick = (playerId, wasAltClick) => {
  const d = dragon;
  if (!d || d.dying > 0 || wasAltClick || d.hitCd > 0) return;
  try {
    const pos = api.getPosition(playerId), dir = api.getPlayerFacingInfo(playerId).dir;
    if (!pos || !dir) return;
    const vx = d.x - pos[0], vy = d.y - (pos[1] + 1.6), vz = d.z - pos[2];
    const t = vx * dir[0] + vy * dir[1] + vz * dir[2];
    if (t < 0 || t > HIT_REACH + HIT_RADIUS) return;
    if (Math.hypot(vx - dir[0] * t, vy - dir[1] * t, vz - dir[2] * t) > HIT_RADIUS) return;

    let dmg = HIT_DAMAGE;
    try {
      const held = api.getHeldItem(playerId);
      if (held && String(held.name).indexOf("Sword") !== -1) dmg = SWORD_DAMAGE;
    } catch (e) {}
    d.hp -= dmg; d.hitCd = HIT_COOLDOWN;
    if (d.hp <= 0) { d.dying = 60; d.killer = playerId; d.ground = groundAt(d.x, d.y, d.z, d.ground); return; }
    const hp = Math.round(d.hp), f = Math.max(0, Math.round(d.hp / MAX_HP * 10));
    api.sendMessage(playerId, "Dragon: " + "█".repeat(f) + "░".repeat(10 - f) + " " + hp + "/" + MAX_HP);
  } catch (e) {}
};

onPlayerChangeBlock = (playerId, x, y, z, fromBlock, toBlock) => {
  try {
    if (isCrystal(toBlock)) { crystals[x + "," + y + "," + z] = [x, y, z]; return; }
    if (!isCrystal(fromBlock)) return;
    delete crystals[x + "," + y + "," + z];
    try { api.broadcastSound(BLAST_SOUND, 1, 1, { x: x + 0.5, y: y + 0.5, z: z + 0.5 }); } catch (e) {}
    for (const pid of api.getPlayerIds()) {
      const p = api.getPosition(pid);
      if (!p) continue;
      const dx = p[0] - (x + 0.5), dy = p[1] - (y + 0.5), dz = p[2] - (z + 0.5);
      const dist = Math.hypot(dx, dy, dz);
      if (dist > BLAST_RADIUS) continue;
      const k = BLAST_KNOCK / Math.max(dist, 1);
      hurt(pid, BLAST_DAMAGE);
      knock(pid, dx * k, 4 + k, dz * k);
    }
  } catch (e) {}
};

onPlayerChat = (playerId, msg) => {
  if (msg === "!dragon") {
    if (dragon) { api.sendMessage(playerId, "The dragon already exists."); return false; }
    dragon = spawnDragon(api.getPosition(playerId));
    api.sendMessage(playerId, dragon ? "The dragon is circling this place!" : "Mesh limit reached, delete some.");
    return false;
  }
  if (msg === "!dragonkill" && dragon) {
    removeDragon();
    api.sendMessage(playerId, "Dragon removed.");
    return false;
  }
};
